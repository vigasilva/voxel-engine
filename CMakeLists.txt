cmake_minimum_required(VERSION 3.20)
project(VoxelEngine VERSION 0.7.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(VOXEL_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(VOXEL_BUILD_APP "Build the SDL2/Vulkan application" ON)
add_library(voxel_core src/elements.c src/composition.c src/material.c src/world.c src/mesh.c src/math3d.c src/position.c src/detail.c src/terrain.c src/scene.c src/terrain_mesh.c src/planet.c src/planet_store.c src/planet_physics.c src/planet_scene.c src/planet_mesh.c src/planet_height.c src/planet_mesh_merge.c src/resources.c src/planet_data.c src/solar.c src/water.c src/surface_texture.c src/planet_mesh_smooth.c src/hydrology.c)
find_package(ZLIB REQUIRED)
target_link_libraries(voxel_core PUBLIC ZLIB::ZLIB)
find_package(SQLite3 REQUIRED)
if(TARGET SQLite3::SQLite3)
  target_link_libraries(voxel_core PUBLIC SQLite3::SQLite3)
else()
  target_link_libraries(voxel_core PUBLIC SQLite::SQLite3)
endif()
target_include_directories(voxel_core PUBLIC src)
if(NOT MSVC)
  target_compile_options(voxel_core PUBLIC -Wall -Wextra -Wpedantic)
  target_link_libraries(voxel_core PUBLIC m)
  if(VOXEL_SANITIZERS)
    target_compile_options(voxel_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(voxel_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(VOXEL_BUILD_APP)
  find_package(Vulkan REQUIRED)
  find_package(SDL2 REQUIRED)
  find_program(GLSLC glslc REQUIRED)
  add_executable(bake_surface_texture tools/bake_surface_texture.c)
  target_link_libraries(bake_surface_texture PRIVATE voxel_core)
  file(GLOB_RECURSE TERRAIN_INPUTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/data/*.edem" "${CMAKE_CURRENT_SOURCE_DIR}/data/*.pdem" "${CMAKE_CURRENT_SOURCE_DIR}/data/*.rivr")
  set(TERRAIN_TEXTURE "${CMAKE_CURRENT_BINARY_DIR}/earth_surface.ptex")
  add_custom_command(OUTPUT "${TERRAIN_TEXTURE}"
    COMMAND bake_surface_texture "${CMAKE_CURRENT_SOURCE_DIR}/data/earth_10min.edem" "${TERRAIN_TEXTURE}.tmp"
    COMMAND ${CMAKE_COMMAND} -E rename "${TERRAIN_TEXTURE}.tmp" "${TERRAIN_TEXTURE}"
    DEPENDS bake_surface_texture ${TERRAIN_INPUTS} VERBATIM)
  add_custom_target(terrain_texture DEPENDS "${TERRAIN_TEXTURE}")
  set(SHADER_OUTPUTS)
  foreach(SHADER voxel.vert voxel.frag hud.vert hud.frag sky.vert sky.frag water.frag occlusion.vert occlusion.frag)
    set(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders/${SHADER}.spv")
    add_custom_command(OUTPUT "${OUTPUT}" COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
      COMMAND ${GLSLC} --target-env=vulkan1.0 "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER}" -o "${OUTPUT}"
      DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/globals.glsl" VERBATIM)
    list(APPEND SHADER_OUTPUTS "${OUTPUT}")
  endforeach()
  add_custom_target(voxel_shaders DEPENDS ${SHADER_OUTPUTS})
  add_executable(voxel_engine src/main.c src/renderer.c src/hud.c src/planet_app.c src/terrain_stream.c src/planet_worker.c)
  add_dependencies(voxel_engine voxel_shaders terrain_texture)
  target_link_libraries(voxel_engine PRIVATE voxel_core SDL2::SDL2 Vulkan::Vulkan)
  target_compile_definitions(voxel_engine PRIVATE VOXEL_SHADER_DIR="${CMAKE_CURRENT_BINARY_DIR}/shaders" VOXEL_TERRAIN_TEXTURE="${TERRAIN_TEXTURE}" VOXEL_EARTH_DATA="${CMAKE_CURRENT_SOURCE_DIR}/data/earth_10min.edem" VOXEL_DETAIL_FETCH="${CMAKE_CURRENT_SOURCE_DIR}/tools/fetch_earth_detail.py")
endif()
add_executable(element_catalog tools/element_catalog.c)
target_link_libraries(element_catalog PRIVATE voxel_core)
include(CTest)

  if(UNIX AND VOXEL_BUILD_APP)
    add_executable(occlusion_seed tests/occlusion_seed.c)
    target_link_libraries(occlusion_seed PRIVATE voxel_core)

  if(UNIX)
    add_executable(terrain_stream_tests tests/terrain_stream_tests.c src/terrain_stream.c)
    target_link_libraries(terrain_stream_tests PRIVATE voxel_core)
    target_compile_definitions(terrain_stream_tests PRIVATE TEST_EARTH_DATA="${CMAKE_CURRENT_SOURCE_DIR}/data/earth_10min.edem" VOXEL_DETAIL_FETCH="${CMAKE_CURRENT_SOURCE_DIR}/tests/terrain_fetch_fixture.py")
    add_test(NAME terrain_download_streaming COMMAND terrain_stream_tests)
